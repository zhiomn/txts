  state.value.matches('inCutscene') ||
  state.value.matches('introCutscene') || // <-- ADICIONADO: Garante que a overlay apareça na intro
  state.value.matches('transitioning') ||
  state.value.matches('fadingOut') ||
  state.value.matches('fadingIn')

<script setup lang="ts">
import { inject, onMounted, onUnmounted, watch, computed } from 'vue';
import type { Emitter } from 'mitt';
import { useDialogueStore } from '@/state/stores/dialogueStore';
import { useTypewriter } from '@/composables/utils/useTypewriter';

const emitter = inject<Emitter<any>>('emitter')!;
const store = useDialogueStore();
const typewriter = useTypewriter();

const advance = () => {
  if (!typewriter.isFinished.value) {
    typewriter.skip(store.getCurrentLine());
    return;
  }
  
  if (!store.next()) {
    store.end();
    emitter.emit('fsm:event', { type: 'DIALOGUE_END' });
  }
};

const onKey = (e: KeyboardEvent) => {
  if (store.isActive && (e.key === 'Enter' || e.key === ' ')) {
    e.preventDefault();
    advance();
  }
};

onMounted(() => window.addEventListener('keydown', onKey));
onUnmounted(() => window.removeEventListener('keydown', onKey));

watch(() => store.currentIndex, () => {
  if (store.isActive) {
    typewriter.start(store.getCurrentLine(), store.getCurrentSpeed());
  }
}, { immediate: true });

const dialogueTextStyle = computed(() => store.dialogueStyle);

import { ref } from 'vue';

const audioCache = new Map<string, HTMLAudioElement>();
const currentMusic = ref<HTMLAudioElement | null>(null);
let audioContextUnlocked = false;

// Exposto para uso interno (como useVoiceManager)
const loadAudio = (src: string): Promise<HTMLAudioElement> => {
  if (audioCache.has(src)) {
    return Promise.resolve(audioCache.get(src)!);
  }
  return new Promise((resolve, reject) => {
    const audio = new Audio(src);
    audio.oncanplaythrough = () => {
      audioCache.set(src, audio);
      resolve(audio);
    };
    audio.onerror = reject;
  });
};

function createAudioManager() {
  const unlockAudio = () => {
    if (audioContextUnlocked) return;
    const silentAudio = new Audio("data:audio/wav;base64,UklGRiQAAABXQVZFZm10IBAAAAABAAEARKwAAIhYAQACABAAZGF0YQAAAAA=");
    silentAudio.play().then(() => {
      audioContextUnlocked = true;
      console.log('[AudioManager] Contexto de áudio desbloqueado com sucesso.');
    }).catch(() => {
      console.warn('[AudioManager] Desbloqueio de áudio aguardando nova interação.');
    });
  };

  const playMusic = async (src: string, loop = true) => {
    console.log(`[DEBUG] 1. playMusic chamada com src: "${src}"`);
    if (currentMusic.value && currentMusic.value.src.endsWith(src)) {
      if (currentMusic.value.paused) currentMusic.value.play();
      return;
    }
    
    if (currentMusic.value) {
      currentMusic.value.pause();
      currentMusic.value.currentTime = 0;
    }

    try {
      console.log(`[DEBUG] 2. Tentando carregar ou obter do cache: "${src}"`);
      const audio = await loadAudio(src);
      audio.loop = loop;
      audio.volume = 0.7;
      console.log('[DEBUG] 3. Áudio carregado. Tentando .play()');
      audio.play();
      currentMusic.value = audio;
    } catch (error) {
      console.error(`[AudioManager] Falha ao tocar música: ${src}`, error);
      console.log('[DEBUG] 4. ERRO no bloco try/catch do playMusic.', error);
    }
  };

  const playSfx = async (src: string, volume = 1) => {
    try {
      const audio = await loadAudio(src);
      audio.currentTime = 0;
      audio.volume = volume;
      audio.play();
    } catch (error) {
      console.error(`[AudioManager] Failed to play SFX: ${src}`, error);
    }
  };

  const stopMusic = () => {
    if (currentMusic.value) {
      currentMusic.value.pause();
      currentMusic.value.currentTime = 0;
      currentMusic.value = null;
    }
  };

  return { playMusic, playSfx, stopMusic, unlockAudio, loadAudio };
}

let instance: ReturnType<typeof createAudioManager>;

export function useAudioManager() {
  if (!instance) {
    instance = createAudioManager();
  }
  return instance;
}

import { useAudioManager } from './useAudioManager';
import type { PlayVoiceCommand } from '@/types/Cutscene';

function createVoiceManager() {
  // Removida a desestruturação de 'loadAudio' para resolver o erro TS6133
  // const { loadAudio } = useAudioManager(); 
  let currentVoiceAudio: HTMLAudioElement | null = null;

  // Função auxiliar para obter o áudio com cache do AudioManager
  const loadAudioFromManager = (src: string) => {
    // Agora acessamos a store para obter a função 'loadAudio'
    const { loadAudio } = useAudioManager(); 
    return loadAudio(src);
  };
  
  const playVoice = async (command: PlayVoiceCommand): Promise<void> => {
    // 1. Para qualquer voz anterior que ainda esteja tocando.
    if (currentVoiceAudio) {
      currentVoiceAudio.pause();
      currentVoiceAudio.currentTime = 0;
    }

    try {
      // 2. Carrega/Obtém do cache o arquivo de áudio.
      const audio = await loadAudioFromManager(command.src);
      
      // 3. Configura e toca.
      audio.loop = false;
      audio.volume = command.volume ?? 1.0;
      audio.currentTime = 0;
      
      currentVoiceAudio = audio;
      audio.play();
      
      // 4. Promessa resolve IMEDIATAMENTE (non-blocking).
      return Promise.resolve();
    } catch (error) {
      console.error(`[VoiceManager] Falha ao tocar áudio de voz: ${command.src}`, error);
      return Promise.resolve(); // Resolva mesmo em caso de erro para não travar a cutscene.
    }
  };

  const stopVoice = () => {
    if (currentVoiceAudio) {
      currentVoiceAudio.pause();
      currentVoiceAudio.currentTime = 0;
      currentVoiceAudio = null;
    }
  };

  return { playVoice, stopVoice };
}

let instance: ReturnType<typeof createVoiceManager>;

export function useVoiceManager() {
  if (!instance) {
    instance = createVoiceManager();
  }
  return instance;
}

import { useEventManager } from '@/composables/events/index';
import { usePlayerStore } from '@/state/stores/playerStore';
import { inject } from 'vue';
import type { Emitter } from 'mitt';

/**
 * Sistema de colisão que verifica se uma posição é bloqueada por eventos sólidos.
 */
function createEventsCollisionSystem() {
  // Precisamos do emitter para passar para o useEventManager
  const emitter = inject<Emitter<any>>('emitter')!;
  const { events } = useEventManager(emitter);
  const playerStore = usePlayerStore();

  /**
   * Verifica se a futura bounding box do jogador colide com algum evento sólido.
   * @param futureX A coordenada X central futura do jogador.
   * @param futureY A coordenada Y central futura do jogador.
   * @returns `true` se a posição for caminhável, `false` se for bloqueada.
   */
  const isEventAreaWalkable = (futureX: number, futureY: number): boolean => {
    const config = playerStore.spriteConfig;

    // Calcula a "caixa de colisão" (bounding box) do jogador na posição futura
    const playerFutureLeft = futureX - config.frameWidth / 2;
    const playerFutureRight = futureX + config.frameWidth / 2;
    const playerFutureTop = futureY - config.frameHeight / 2;
    const playerFutureBottom = futureY + config.frameHeight / 2;

    // Filtra apenas os eventos que são sólidos
    const solidEvents = events.value.filter(e => e.triggerArea.isSolid);

    for (const evt of solidEvents) {
      const area = evt.triggerArea;

      // Verifica se a caixa do jogador colide com a área do evento sólido
      if (
        playerFutureRight > area.x &&
        playerFutureLeft < area.x + area.width &&
        playerFutureBottom > area.y &&
        playerFutureTop < area.y + area.height
      ) {
        return false; // Colisão detectada, área não é caminhável
      }
    }

    return true; // Nenhuma colisão encontrada, área é caminhável
  };

  return { isEventAreaWalkable };
}

let instance: ReturnType<typeof createEventsCollisionSystem>;

export function useCollisionEvents() {
  if (!instance) {
    instance = createEventsCollisionSystem();
  }
  return instance;
}

import { useWorldMapManager } from '@/composables/useWorldMapManager';
import { useGameConfig } from '@/composables/useGameConfig';

/**
 * Sistema de colisão que verifica se uma posição é caminhável com base nos dados do tilemap.
 */
function createTilemapCollisionSystem() {
  const { collisionData } = useWorldMapManager();
  const { config: gameConfig } = useGameConfig();

  const isTileWalkable = (x: number, y: number): boolean => {
    // Se não houver dados de colisão definidos para este mapa...
    if (!collisionData.value || collisionData.value.data.length === 0) {
      // ...então, do ponto de vista do tilemap, tudo é caminhável.
      return true; // <-- CORREÇÃO: Trata dados nulos OU array de dados vazio como caminhável
    }

    const { width, data } = collisionData.value;
    const tileX = Math.floor(x / gameConfig.tileSize);
    const tileY = Math.floor(y / gameConfig.tileSize);

    const index = tileY * width + tileX;

    if (index < 0 || index >= data.length) {
      // Fora dos limites do mapa de colisão
      return false;
    }

    // 0 = caminhável, qualquer outro valor = bloqueado
    return data[index] === 0;
  };

  return { isTileWalkable };
}

let instance: ReturnType<typeof createTilemapCollisionSystem>;

export function useCollisionTilemap() {
  if (!instance) {
    instance = createTilemapCollisionSystem();
  }
  return instance;
}

      });
    }
  
    if (newState === 'roaming' && (oldState === 'inDialogue' || oldState === 'inCutscene')) {
      checkTriggers();
    }
  }, { immediate: true });
  
  watch(state, async (currentState) => {
    if (currentState.matches('fadingIn') && currentState.context.mapToLoad) {
      gameStore.setIsReadyForRender(false);
      
      setCurrentMap(currentState.context.mapToLoad);
      await loadEvents(currentState.context.mapToLoad);
      
      if (mapData.value) {
        sfxStore.setMapSfx(mapData.value.sfx);
        if (mapData.value.playerSpriteConfig) {
          playerStore.setSpriteConfig(mapData.value.playerSpriteConfig);
        }
      }
  
      if (currentState.context.nextPosition) {
        const { x, y, direction } = currentState.context.nextPosition;
        playerStore.setPosition(x, y);
        playerStore.setDirection(direction);
      } else {
        const devPosition = devMode.getPlayerStartPosition();
        if (devPosition && devMode.isEnabled()) {
          playerStore.setPosition(devPosition.x, devPosition.y);
          playerStore.setDirection(devPosition.direction);
        } else if (mapData.value) { 
          const { x, y, direction } = mapData.value.startPosition;
          playerStore.setPosition(x, y);
          playerStore.setDirection(direction);
        }
      }
      
      send({ type: 'MAP_LOADED' });
      
      gameStore.setIsReadyForRender(true);
      await nextTick();
  
      if (mobileShellContainer.value) {
        gsap.fromTo(mobileShellContainer.value, { opacity: 0 }, { opacity: 1, duration: 1.0 });
      }
    }
    
    if ((currentState.matches('inCutscene') || currentState.matches('introCutscene') || currentState.matches('transitioning')) && currentState.context.currentCutscene) {
      playCutscene(currentState.context.currentCutscene);
    }
  }, { deep: true, immediate: true });
}

import { reactive } from 'vue';
import { useWorldMapManager } from '../useWorldMapManager';
import { useDataService } from '../data';

const state = reactive({
  progress: 0,
  total: 0,
  isLoaded: false,
});

const loadImage = (src: string) => new Promise((resolve, reject) => {
  const img = new Image();
  img.onload = resolve;
  img.onerror = reject;
  img.src = src;
});

const loadAudio = (src: string) => new Promise<void>((resolve, reject) => {
  const audio = new Audio();
  audio.onloadeddata = () => resolve();
  audio.onerror = reject;
  audio.src = src;
});

const loadJSON = (src: string) => fetch(src).then(res => res.json());

async function loadAssets(emitter: any) {
  const { loadMapAndCollision } = useWorldMapManager();
  const dataService = useDataService();
  
  try {
    // Load manifest first
    const manifest = await loadJSON('./data/_manifest.json');
    
    const imagePromises = (manifest.images || []).map(loadImage);
    const audioPromises = (manifest.audio || []).map((a: any) => loadAudio(a.src));
    const mapPromises = (manifest.maps || []).map((mapPath: string) => loadMapAndCollision(mapPath));
    
    const allPromises = [...imagePromises, ...audioPromises, ...mapPromises];
    state.total = allPromises.length;

    allPromises.forEach(promise => {
      promise.then(() => {
        state.progress++;
      });
    });

    await Promise.all(allPromises);

    // Após os assets, inicializa o serviço de dados do jogo
    await dataService.initialize();

    state.isLoaded = true;
    console.log('[AssetLoader] All assets loaded.');
    emitter.emit('fsm:event', { type: 'ASSETS_LOADED' });

  } catch (error) {
    console.error('[AssetLoader] CRITICAL ERROR DURING ASSET LOADING:', error);
  }
}

export function useAssetLoader() {
  return {
    loadingState: state,
    loadAssets,
  };
}

import { onMounted, onUnmounted } from 'vue';
import { useMovementSystem } from '../input/index_movement'; // <-- CORRIGIDO

let animationFrameId: number;

function createGameLoop() {
  const { update: updateMovement } = useMovementSystem(); // <-- MUDANÇA
  let lastTime = 0;

  const loop = (timestamp: number) => {
    if (lastTime === 0) {
      lastTime = timestamp;
    }
    const deltaTime = (timestamp - lastTime) / 1000;
    lastTime = timestamp;

    updateMovement(deltaTime); // <-- MUDANÇA
    
    animationFrameId = requestAnimationFrame(loop);
  };

  const start = () => {
    if (!animationFrameId) {
      lastTime = 0;
      animationFrameId = requestAnimationFrame(loop);
    }
  };

  const stop = () => {
    if (animationFrameId) {
      cancelAnimationFrame(animationFrameId);
      animationFrameId = 0;
    }
  };

  return { start, stop };
}

let instance: ReturnType<typeof createGameLoop>;

export function useGameLoop() {
  if (!instance) {
    instance = createGameLoop();
  }

  onMounted(instance.start);
  onUnmounted(instance.stop);

  return instance;
}

import { reactive } from 'vue';
import type { GameEventLogic } from '@/types/Event';
import type { CutsceneDefinition } from '@/types/Cutscene';

// Armazenamento genérico para assets de dados (Map<Tipo, Map<ID, Objeto>>)
const dataCache = reactive(new Map<string, Map<string, any>>());

/**
 * Inicializa um tipo de asset no cache se não existir.
 */
function ensureCacheType(type: string) {
  if (!dataCache.has(type)) {
    dataCache.set(type, new Map());
  }
  return dataCache.get(type)!;
}

/**
 * Carrega um arquivo barrel e processa cada item listado.
 * @param barrelPath Caminho para o arquivo JSON barrel.
 * @param itemPrefix Prefixo do caminho para os arquivos de item individuais.
 * @param type O tipo de dado sendo carregado (ex: 'characters', 'cards').
 */
async function loadAndProcessBarrel(barrelPath: string, itemPrefix: string, type: string): Promise<void> {
  try {
    const barrelResponse = await fetch(barrelPath);
    const barrelFiles: string[] = await barrelResponse.json();
    const cacheMap = ensureCacheType(type);

    const itemPromises = barrelFiles.map(async (fileName) => {
      const itemPath = `${itemPrefix}${fileName}`;
      const itemResponse = await fetch(itemPath);
      const itemData = await itemResponse.json();
      if (itemData && itemData.id) {
        cacheMap.set(itemData.id, itemData);
      }
    });

    await Promise.all(itemPromises);
    console.log(`[DataService] Barrel '${type}' carregado com sucesso.`);
  } catch (error) {
    console.error(`[DataService] Falha ao carregar o arquivo barrel ${barrelPath}:`, error);
  }
}

function createDataService() {
  /**
   * Inicializa o serviço, carregando todos os barrels definidos.
   */
  const initialize = async () => {
    // Carrega Personagens
    await loadAndProcessBarrel('./data/characters.json', './data/characters/', 'characters');
    // Carrega Cartas
    await loadAndProcessBarrel('./data/cards.json', './data/cards/', 'cards');
    // Carrega Eventos
    await loadAndProcessBarrel('./data/events.json', './data/events/', 'events');
  };

  /**
   * Obtém um item do cache pelo tipo e ID.
   */
  const getItem = (type: string, id: string): any => {
    return dataCache.get(type)?.get(id);
  };
  
  /**
   * NOVO: Carrega a definição de uma cutscene sob demanda.
   */
  const getCutsceneById = async (id: string): Promise<CutsceneDefinition | null> => {
    const cutsceneUrl = `./data/cutscenes/${id}.json`;
    try {
      const response = await fetch(cutsceneUrl);
      if (!response.ok) throw new Error(`Status ${response.status}`);
      return await response.json();
    } catch (error) {
      console.error(`[DataService] Falha ao carregar cutscene ${id}:`, error);
      return null;
    }
  };

  /**
   * NOVO: Carrega a configuração de sorteio de cartas sob demanda.
   */
  const getDrawConfigBySource = async (source: string): Promise<any | null> => {
    try {
      const response = await fetch(source);
      if (!response.ok) throw new Error(`Status ${response.status}`);
      return await response.json();
    } catch (error) {
      console.error(`[DataService] Falha ao carregar configuração de sorteio: ${source}`, error);
      return null;
    }
  };

  return {
    initialize,
    getItem,
    getCutsceneById,
    getDrawConfigBySource,
  };
}

let instance: ReturnType<typeof createDataService>;

export function useDataService() {
  if (!instance) {
    instance = createDataService();
  }
  return instance;
}

import { useSwitchesStore } from '@/state/stores/switchesStore';
import type { EventCondition } from '@/types/Event';
import { logConditionFailure } from '@/composables/utils/logging/logConditions';

export function checkConditions(conditions: EventCondition[]): boolean {
  const switchesStore = useSwitchesStore();

  if (!conditions || conditions.length === 0) return true;

  for (const condition of conditions) {
    const switchValue = switchesStore.getSwitch(condition.switchId) ?? false;
    
    if (switchValue !== condition.value) {
      logConditionFailure(condition.switchId, condition.value, switchValue);
      return false; // Condição falhou
    }
  }
  return true; // Todas as condições foram satisfeitas
}

/**
 * Módulo de responsabilidade única para despachar a ação de um GameEvent
 * como um evento Mitt tipado para os ouvintes do jogo (listeners/).
 */
export function executeEvent(emitter: Emitter<any>, evt: GameEvent) {
  const { executeSetSwitch } = useSwitchActions();
  const dialogueStore = useDialogueStore(); // NOVO

  console.log(`[EventExecutor] Despachando ação: ${evt.action.type} para ${evt.id}`);

  if (evt.action.type === 'dialogue') {
    // CORREÇÃO CRÍTICA: Trava de Diálogo
    if (dialogueStore.isActive) {
        console.warn(`[EventExecutor] Diálogo para ${evt.id} BLOQUEADO: outro diálogo já está ativo.`);
        return;
    }
    emitter.emit('event:dialogue-triggered', evt.action);
  } else if (evt.action.type === 'cutscene') {
    emitter.emit('event:cutscene-triggered', evt.action);
  } else if (evt.action.type === 'cardDraw') {
    emitter.emit('event:card-draw-triggered', evt.action);
  } else if (evt.action.type === 'playVoice') {
    emitter.emit('event:voice-triggered', evt.action);
  } else if (evt.action.type === 'setSwitch') {
    executeSetSwitch(evt.action);
  }
}

import { useEventState } from './eventState';
import { useWorldMapManager } from '@/composables/useWorldMapManager';
import { getEventLogicById } from '@/composables/data/data_access';
import type { GameEvent, GameEventLogic } from '@/types/Event';

export function useEventLoader() {
  const { setEvents } = useEventState();
  const { mapData } = useWorldMapManager();

  async function loadEvents(mapId: string): Promise<void> {
    if (!mapData.value) {
      console.error('[EventLoader] Map data not available.');
      setEvents([]);
      return;
    }

    const finalEvents: GameEvent[] = [];
    
    // 1. Iterar sobre os placements (posições) definidos no mapa
    for (const placement of mapData.value.eventPlacements) {
      const logic = getEventLogicById(placement.eventId);
      
      if (!logic) {
        console.warn(`[EventLoader] Lógica de evento não encontrada para ID: ${placement.eventId}`);
        continue;
      }
      
      // 2. Mesclar a lógica (triggerAreaBase, action, conditions) com a posição (x, y)
      const finalEvent: GameEvent = {
        ...logic,
        triggerArea: {
          ...logic.triggerAreaBase,
          x: placement.x,
          y: placement.y,
        }
      };
      
      finalEvents.push(finalEvent);
    }

    setEvents(finalEvents);
    console.log(`[EventLoader] ${finalEvents.length} eventos montados para o mapa.`);
  }

  return { loadEvents };
}

import type { Emitter } from 'mitt';
import type { GameEvent } from '@/types/Event';
import { usePlayerStore } from '@/state/stores/playerStore';
import { useEventState } from './eventState';
import { checkConditions } from './eventConditions';
import { executeEvent } from './eventExecutor'; 
import { useGameStore } from '@/state/stores/gameStore';

export function useEventTriggers(emitter: Emitter<any>) {
  const playerStore = usePlayerStore();
  const { events } = useEventState();
  const gameStore = useGameStore();

  function checkTriggers() {
    // CORREÇÃO CRÍTICA: Não dispara eventos se o jogo não estiver em roaming
    if (gameStore.currentState !== 'roaming') {
        return;
    }
    
    console.log(`[EventTriggers] Verificando gatilhos no estado: ${gameStore.currentState}`);

    const p = playerStore.position;
    const config = playerStore.spriteConfig;
    
    const playerLeft = p.x - config.frameWidth / 2;
    const playerRight = p.x + config.frameWidth / 2;
    const playerTop = p.y - config.frameHeight / 2;
    const playerBottom = p.y + config.frameHeight / 2;

    for (const evt of events.value) {
      if (evt.triggerArea.isSolid) continue;
      
      const area = evt.triggerArea;
      
      // 1. Checagem de Colisão (simples)
      if (
        playerRight > area.x &&
        playerLeft < area.x + area.width &&
        playerBottom > area.y &&
        playerTop < area.y + area.height
      ) {
        console.log(`[EventTriggers] COLISÃO DETECTADA com ${evt.id}.`);
        
        // 2. Verifica Condições
        if (!evt.conditions || checkConditions(evt.conditions)) {
          console.log(`[EventTriggers] EXECUTANDO AÇÃO para ${evt.id} (Tipo: ${evt.action.type}).`);
          executeEvent(emitter, evt); 
        } else {
          console.log(`[EventTriggers] Condições de ${evt.id} FALHARAM. Ação não executada.`);
        }
      }
    }
  }

  return { checkTriggers, events };
}

/**
 * Composable para filtrar a lista de eventos do mapa, retornando apenas
 * aqueles que devem ser visíveis e acionáveis no estado atual do jogo.
 */
export function useVisibleEvents() {
  const emitter = inject<Emitter<any>>('emitter')!;
  const { events } = useEventManager(emitter);

  const visibleEvents = computed<GameEvent[]>(() => {
    const filtered = events.value.filter(evt => {
      // 1. Eventos Sólidos (invisíveis) são sempre "visíveis" para a lógica de colisão
      if (evt.triggerArea.isSolid) return true;
      
      // 2. Se não tiver sprite, não há o que renderizar
      if (!evt.triggerArea.sprite) return false; 

      // 3. Se não houver condições, o evento é sempre visível
      if (!evt.conditions || evt.conditions.length === 0) {
        console.log(`[VisibleEvents] Evento ${evt.id} VISÍVEL (Sem Condições).`);
        return true;
      }

      // 4. Se houver condições, verificar o estado atual
      // A função checkConditions usa a useSwitchesStore internamente.
      const isVisible = checkConditions(evt.conditions);
      if (isVisible) {
        console.log(`[VisibleEvents] Evento ${evt.id} VISÍVEL (Condições OK).`);
      } else {
        console.log(`[VisibleEvents] Evento ${evt.id} INVISÍVEL (Condições FALHARAM).`);
      }
      return isVisible;
    });
    
    console.log(`[VisibleEvents] Total de eventos renderizáveis: ${filtered.length}`);
    return filtered;
  });

  return { visibleEvents };
}

import { gsap } from 'gsap';

// Função utilitária para aguardar a população do array de refs do Vue
const waitForRefs = (refs: any[], expectedCount: number): Promise<void> => {
    return new Promise(resolve => {
        const check = () => {
            // Conta quantos refs são válidos e possuem um elemento DOM
            const validCount = refs.filter(r => r && r.$el).length;
            
            if (validCount >= expectedCount) {
                // Se a contagem esperada foi atingida, resolve
                resolve();
            } else {
                // Senão, tenta novamente no próximo frame de animação
                requestAnimationFrame(check);
            }
        };
        // Inicia a verificação no próximo tick para dar tempo ao VUE
        requestAnimationFrame(check);
    });
};

export function useCardDrawAnimator() {

    // Adiciona o expectedCount para saber quando o VUE terminou de renderizar
    const animateCardAppearance = async (cardRefs: any[], expectedCount: number): Promise<void> => {
        // 1. Aguarda até que todos os refs sejam populados
        if (expectedCount > 0) {
            await waitForRefs(cardRefs, expectedCount);
        } else {
            console.warn('[CardDrawAnimator] Expected count é zero, ignorando animação de aparição.');
            return;
        }

        return new Promise(resolve => {
            // 2. Garante que estamos no próximo tick após o waitForRefs
            nextTick(() => { 
                const validTargets = cardRefs
                    .filter(comp => comp && comp.$el)
                    .map(comp => comp.$el);
                
                if (validTargets.length > 0) {
                    gsap.to(
                        validTargets, 
                        {
                            opacity: 1,
                            scale: 1,
                            duration: CARD_APPEAR_DURATION,
                            stagger: CARD_APPEAR_STAGGER,
                            ease: 'power2.out',
                            onComplete: resolve
                        }
                    );
                } else {
                    console.error('[CardDrawAnimator] Alvos válidos não encontrados para animação de aparição após espera.');
                    resolve();
                }
            });
        });
    };

    const animateCardReveal = (chosenCard: any, otherCards: any[], outcome: any): Promise<void> => {
        const otherValidTargets = otherCards
            .filter(c => c && c.$el)
            .map(c => c.$el);

        return new Promise(resolve => {
            if (!chosenCard || !chosenCard.$el) {
                console.error('[CardDrawAnimator] CRÍTICO: chosenCard.$el é inválido. Animação abortada.');
                return resolve();
            }

            const tl = gsap.timeline({
                onComplete: resolve // Resolve a promise quando a timeline terminar
            });

            tl
            .to(chosenCard.$el.querySelector('.card-inner'), {
                rotateY: 180,
                duration: CARD_FLIP_DURATION,
                ease: 'power2.inOut'
            })
            .to(otherValidTargets, {
                opacity: 0,
                scale: 0.9,
                duration: CARD_HIDE_OTHERS_DURATION,
                stagger: 0.1,
                ease: 'power1.in'
            }, "-=0.2")
            .to({}, { duration: CARD_RESULT_VIEW_DURATION }) // Pausa para visualização
            .to(chosenCard.$el, {
                opacity: 0,
                scale: 0.9,
                duration: CARD_HIDE_CHOSEN_DURATION,
                ease: 'power1.in'
            });
        });
    };

    return {
        animateCardAppearance,
        animateCardReveal,
    };
}

interface CardRef { id: string; weight: number; }
interface DrawConfig { cardCount: number; cards: CardRef[]; }

export function useCardDrawLogic(config: Ref<DrawConfig | null>, emitter: Emitter<any>, animateReveal: (chosen: any, others: any, outcome: any) => Promise<void>) {
    const { draw } = useLottery();

    const assembleCardData = (cardRef: CardRef) => {

            return null;
        }
        return {
            ...staticData,
            weight: cardRef.weight
        };
    };

    // 1. Array de objetos de slot para o v-for (renderização)
    const cardsForRender = computed(() => {
        if (!config.value || !config.value.cards || config.value.cardCount === 0) return [];
        
        return Array.from({ length: config.value.cardCount }, (_, index) => {
            // Garante que o slot use o ID da carta correta (e.g., Ovelha é sempre o índice 0)
            const cardRefIndex = Math.min(index, config.value!.cards.length - 1);
            
            return {
                index, // Índice de renderização real (0 ou 1)
                id: config.value!.cards[cardRefIndex].id
            };
        });
    });
    
    const getFinalOutcomes = computed(() => {
        if (!config.value || !config.value.cards) return [];
        
        const outcomes = config.value.cards
            .map(assembleCardData)
            .filter(c => c !== null);
            
        // 2. Filtragem de Cartas Já Escolhidas
        const possibleOutcomes = outcomes.filter((o: any) => {
            // Se for um sorteio de carta única (ex: ovelha), não filtra.
            if (config.value!.cardCount === 1) return true;

            const switchKey = `${o.id}_chosen`;
            const hasBeenChosen = switchesStore.getSwitch(switchKey);
            return !hasBeenChosen;
        });
        
        return possibleOutcomes;
    });

    const chooseCard = async (index: number, cardRefs: any[]) => {
        if (interactionDisabled.value) {
            console.warn('[CardDrawLogic] Interação desabilitada.');
            return;
        }
        interactionDisabled.value = true;
        
        const possibleOutcomes = getFinalOutcomes.value;

        if (possibleOutcomes.length === 0) {
            logLogicError('Sem opções válidas para sorteio, travando.'); // <-- USO DO LOG REUTILIZÁVEL
            interactionDisabled.value = false;
            return;
        }

        // 3. Executa o sorteio
        const outcome = draw(possibleOutcomes);
        drawnOutcome.value = outcome;
        chosenCardIndex.value = index;

        // 4. Salva o switch
        switchesStore.setSwitch(`${outcome.id}_chosen`, true);
        
        // 5. Orquestra a Animação e Ação Final
        await nextTick();
        const chosenCard = cardRefs[index];
        const otherCards = cardRefs.filter((_, i) => i !== index);

        if (!chosenCard || !chosenCard.$el) {
            logLogicError('CRÍTICO: chosenCard.$el é inválido, não é possível animar.'); // <-- USO DO LOG REUTILIZÁVEL
            interactionDisabled.value = false; // Tenta recuperar a interação
            return;
        }
        
        try {
             // O await aqui é crucial para a ordem dos eventos
            await animateReveal(chosenCard, otherCards, outcome); 
        } catch (error) {
            console.error('[CardDrawLogic] Erro durante a animação de revelação:', error);
        }

        // 6. Envia o evento de conclusão
        if (outcome?.action) {
            emitter.emit('fsm:event', { type: 'CARD_CHOSEN', action: outcome.action });
        }
    };

    return {
        drawnOutcome,
        cardsForRender,
        interactionDisabled,
        chosenCardIndex,
        chooseCard,
    };
}

/**
 * Um sistema puro e reutilizável para realizar sorteios ponderados.
 */
interface Outcome {
  weight: number;
  [key: string]: any;
}

function createLotterySystem() {
  /**
   * Sorteia um resultado de uma lista de possibilidades com base em seus pesos.
   * @param outcomes Um array de objetos, onde cada objeto deve ter uma propriedade 'weight'.
   * @returns O objeto do resultado sorteado, ou null se a lista estiver vazia.
   */
  const draw = <T extends Outcome>(outcomes: T[]): T | null => {
    if (!outcomes || outcomes.length === 0) {
      return null;
    }

    const totalWeight = outcomes.reduce((sum, outcome) => sum + outcome.weight, 0);
    let random = Math.random() * totalWeight;

    for (const outcome of outcomes) {
      if (random < outcome.weight) {
        return outcome;
      }
      random -= outcome.weight;
    }

    // Fallback caso algo dê errado com os cálculos de ponto flutuante
    return outcomes[outcomes.length - 1];
  };

  return { draw };
}

let instance: ReturnType<typeof createLotterySystem>;

export function useLottery() {
  if (!instance) {
    instance = createLotterySystem();
  }
  return instance;
}

import { inject } from 'vue';
import type { Emitter } from 'mitt';
import { usePlayerStore } from '@/state/stores/playerStore';
import { useCollisionTilemap, useCollisionEvents } from '../collision/index_collision';

const PLAYER_SPEED = 420; // pixels per second

/**
 * Sistema que executa o movimento do jogador, aplicando física,
 * verificando colisões e emitindo eventos.
 */
function createMovementCoreSystem() {
  const playerStore = usePlayerStore();
  const { isTileWalkable } = useCollisionTilemap();
  const { isEventAreaWalkable } = useCollisionEvents();
  const emitter = inject<Emitter<any>>('emitter')!;
  let stepSoundCooldown = 0;

  const executeMovement = (intent: { x: number }, deltaTime: number) => {
    playerStore.setIsWalking(true);
    
    if (intent.x > 0) playerStore.setDirection('right');
    if (intent.x < 0) playerStore.setDirection('left');
    
    const newX = playerStore.position.x + intent.x * PLAYER_SPEED * deltaTime;
    const newY = playerStore.position.y;

    // 1. Verifica colisão com eventos sólidos
    if (!isEventAreaWalkable(newX, newY)) {
      return;
    }

    // 2. Verifica colisão com o tilemap
    const config = playerStore.spriteConfig;
    const halfWidth = config.frameWidth / 2;
    let leadingEdgeX = newX;
    if (playerStore.direction === 'right') leadingEdgeX += halfWidth;
    if (playerStore.direction === 'left') leadingEdgeX -= halfWidth;
    
    if (!isTileWalkable(leadingEdgeX, newY + 16)) {
      return;
    }
    
    // Movimento permitido
    playerStore.setPosition(newX, newY);

    stepSoundCooldown -= deltaTime;
    if (stepSoundCooldown <= 0) {
      emitter.emit('player:step');
      stepSoundCooldown = 1.8;
    }
  };

  const stopMovement = () => {
    playerStore.setIsWalking(false);
  };

  return { executeMovement, stopMovement };
}

let instance: ReturnType<typeof createMovementCoreSystem>;

export function useMovementCore() {
  if (!instance) {
    instance = createMovementCoreSystem();
  }
  return instance;
}

/**
 * Sistema que lê a entrada do jogador e o estado do jogo para determinar
 * a intenção de movimento.
 */
function createMovementInputSystem() {
  const { inputState } = useHorizontalInput();
  const gameStore = useGameStore();

  /**
   * Retorna a intenção de movimento do jogador.
   * @returns Um objeto com a direção (ex: { x: -1 }) ou null se não houver movimento.
   */
  const getMovementIntent = (): { x: number } | null => {
    if (gameStore.currentState !== 'roaming') {
      return null;
    }

    let dx = 0;
    if (inputState.left) dx -= 1;
    if (inputState.right) dx += 1;
    
    if (dx === 0) {
      return null;
    }

    return { x: dx };
  };

  return { getMovementIntent };
}

  }

import { reactive, onMounted, onUnmounted } from 'vue';

const state = reactive({
  left: false,
  right: false,
});

const keyMap: Record<string, keyof typeof state> = {
  a: 'left',
  arrowleft: 'left',
  d: 'right',
  arrowright: 'right',
};

const handleKeyDown = (event: KeyboardEvent) => {
  const key = event.key.toLowerCase();
  const action = keyMap[key];
  if (action) {
    event.preventDefault();
    state[action] = true;
  }
};

const handleKeyUp = (event: KeyboardEvent) => {
  const key = event.key.toLowerCase();
  const action = keyMap[key];
  if (action) {
    state[action] = false;
  }
};

// Funções públicas para controle por toque
const setLeft = (value: boolean) => { state.left = value; };
const setRight = (value: boolean) => { state.right = value; };

let isInitialized = false;

export function useHorizontalInput() {
  if (!isInitialized) {
    onMounted(() => {
      window.addEventListener('keydown', handleKeyDown);
      window.addEventListener('keyup', handleKeyUp);
    });
    onUnmounted(() => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('keyup', handleKeyUp);
    });
    isInitialized = true;
  }

  return { inputState: state, setLeft, setRight };
}

/**
 * Carrega e traduz chaves de áudio de voz a partir de um arquivo de localização de áudio.
 * @param key A chave da tradução (ex: "voice_burro_poem").
 * @param locale O código do idioma.
 * @returns Uma promessa que resolve para o caminho do arquivo de áudio (URL).
 */
async function translate(key: string, locale: string): Promise<string> {
  // Neste projeto simplificado, usaremos um único arquivo de voz: voice.json
  const filePath = `./data/locales/${locale}/audio/voice.json`;

  let translations = cache.get(filePath);

  if (!translations) {
    try {
      translations = await fetch(filePath).then(res => res.json());
      cache.set(filePath, translations!);
    } catch (error) {
      console.error(`[Localization] Falha ao carregar arquivo de áudio: ${filePath}`, error);
      cache.set(filePath, {}); // Evita futuras tentativas falhas
      return key;
    }
  }

  return translations?.[key] || key;
}

  };
}

// --- Singleton Pattern ---
// Isso garante que tenhamos apenas uma instância do nosso sistema de localização em toda a aplicação.

let instance: ReturnType<typeof createLocalizationSystem>;

function createLocalizationSystem() {
  // --- Estado Central ---
  const currentLocale = ref('pt-BR');

  // --- Módulos Internos ---
  const ui = createInternalLocalizationUI();
  const events = createInternalLocalizationEvents();
  const cutscenes = createInternalLocalizationCutscene();
  const audio = createInternalLocalizationAudio(); // <--- NOVO

  // --- API Pública ---

  /**
   * Define o idioma atual para toda a aplicação.
   * Isso carregará as traduções globais da UI e limpará os caches
   * das traduções contextuais para que sejam recarregadas no novo idioma.
   * @param locale O código do idioma (ex: "pt-BR").
   */
  async function setLocale(locale: string) {
    currentLocale.value = locale;
    // Notifica cada módulo da mudança
    await ui.load(locale);
    events.clearCache();
    cutscenes.clearCache();
    audio.clearCache(); // <--- NOVO
  }

  /**
   * Obtém uma tradução do arquivo global da UI. A operação é síncrona.
   * @param key A chave da tradução (ex: "splashScreen.playButton").
   * @returns A string traduzida ou a chave como fallback.
   */
  const t_ui = (key: string) => ui.translate(key);

  /**
   * Obtém uma tradução de um arquivo de evento. A operação é assíncrona
   * e carrega/cacheia o arquivo de evento necessário sob demanda.
   * @param key A chave da tradução (ex: "CAMPO_espantalho1_line1").
   * @returns Uma promessa que resolve para a string traduzida.
   */
  const t_event = (key: string) => events.translate(key, currentLocale.value);

  /**
   * Obtém uma tradução de um arquivo de cutscene. A operação é assíncrona
   * e carrega/cacheia o arquivo de cutscene necessário sob demanda.
   * @param key A chave da tradução (ex: "burro_poem").
   * @returns Uma promessa que resolve para a string traduzida.
   */
  const t_cutscene = (key: string) => cutscenes.translate(key, currentLocale.value);
  
  /**
   * Obtém o URL de um arquivo de áudio de voz. A operação é assíncrona.
   * @param key A chave do áudio (ex: "voice_burro_poem").
   * @returns Uma promessa que resolve para o URL do arquivo de áudio.
   */
  const t_voiceAudio = (key: string) => audio.translate(key, currentLocale.value); // <--- NOVO
  
  return {
    setLocale,
    t_ui,
    t_event,
    t_cutscene,
    t_voiceAudio, // <--- NOVO
    // Expõe o locale como readonly para que componentes possam reagir a mudanças de idioma
    locale: readonly(currentLocale),
  };
}

/**
 * O composable público para acessar o sistema de internacionalização.
 */
export function useLocalization() {
  if (!instance) {
    instance = createLocalizationSystem();
  }
  return instance;
}

import { ref } from 'vue';

const uiTranslations = ref<Record<string, any>>({});

// Lista de módulos de UI para carregar
const UI_MODULES = ['loading', 'splash', 'card', 'extras'];

async function load(locale: string) {
  try {
    const promises = UI_MODULES.map(async (module) => {
      const path = `./data/locales/${locale}/ui/${module}.json`;
      const res = await fetch(path);
      return res.json();
    });

    const loadedModules = await Promise.all(promises);

    // Mescla todos os módulos em um único objeto de tradução
    uiTranslations.value = loadedModules.reduce((acc, curr) => ({ ...acc, ...curr }), {});
    
    console.log(`[Localization] Módulos de UI carregados para: ${locale}`);

  } catch (error) {
    console.error(`[Localization] Falha ao carregar módulos de UI para o locale: ${locale}`, error);
    uiTranslations.value = {}; 
  }
}

/**
 * Função auxiliar para navegar em objetos aninhados com segurança.
 */
function getProp(obj: any, path: string, fallback: string): string {
  if (!obj) return fallback;
  return path.split('.').reduce((acc, part) => (acc && acc[part] !== undefined) ? acc[part] : undefined, obj) || fallback;
}

function translate(key: string): string {
  if (!key) return '';
  return getProp(uiTranslations.value, key, key);
}

export function createInternalLocalizationUI() {
  return {
    load,
    translate,
  };
}

import { useCutsceneUIStore } from '@/state/stores/cutsceneUIStore';
import type { CutsceneDefinition } from '@/types/Cutscene';
import { executeDialogue } from './cutscene/dialogue';
import { executeHideActor } from './cutscene/hideActor';

import { executePlaySfx } from './cutscene/playSfx';
import { executeStopMusic } from './cutscene/stopMusic';
import { executePlayVoice } from './cutscene/playVoice';
import { getCutsceneById } from './data'; // IMPORTAÇÃO ATUALIZADA

const commandExecutors = {
  dialogue: executeDialogue,
  hideActor: executeHideActor,
  showActor: executeShowActor,
  wait: executeWait,
  playMusic: executePlayMusic,
  playSfx: executePlaySfx,
  stopMusic: executeStopMusic,
  playVoice: executePlayVoice,
};

function createCutscenePlayer(emitter: any) {
  const uiStore = useCutsceneUIStore();

  const play = async (cutsceneId: string) => {
    try {
      // ATUALIZADO: Usa o DataService para carregar a cutscene
      const cutscene: CutsceneDefinition | null = await getCutsceneById(cutsceneId);
      
      if (!cutscene || !cutscene.sequence) throw new Error(`Invalid cutscene format: ${cutsceneId}`);
      
      const bgConfig = cutscene.background || {};
      const fadeInDuration = bgConfig.fadeIn || 0;
      const fadeOutDuration = bgConfig.fadeOut || 0;
      
      uiStore.showOverlay();
      uiStore.setBackground({ color: bgConfig.color, image: bgConfig.image });

      await new Promise(resolve => {
        uiStore.setOverlayOpacity(1);
        setTimeout(resolve, fadeInDuration);
      });

      for (const command of cutscene.sequence) {
        const executor = commandExecutors[command.type as keyof typeof commandExecutors];
        if (executor) {
          await executor(command as any);
        } else {
          console.warn(`[CutscenePlayer] Comando desconhecido: ${command.type}`);
        }
      }
      
      await new Promise(resolve => {
        uiStore.setOverlayOpacity(0);
        setTimeout(resolve, fadeOutDuration);
      });

      uiStore.reset();
      
      emitter.emit('fsm:event', { type: 'CUTSCENE_END', onEnd: cutscene.onEnd });

    } catch (error) {
      console.error(`[CutscenePlayer] Failed to play cutscene ${cutsceneId}:`, error);
      uiStore.reset();
      emitter.emit('fsm:event', { type: 'CUTSCENE_END' });
    }
  };

  return { play };
}

export function useCutscenePlayer(emitter: any) {
  return createCutscenePlayer(emitter);
}

      console.log('[GameConfig] Configuration loaded.');
    } catch (error) {
      console.error('[GameConfig] Failed to load game configuration, using defaults.', error);
    }
  }

  return {
    config: readonly(config),
    loadConfig,
  };
}

import { ref, computed } from 'vue';
import type { MapDefinition, MapManifest, EventPlacement } from '@/types/MapDefinition';

const mapCache = new Map<string, MapDefinition>();
const collisionCache = new Map<string, { width: number; height: number; data: number[] }>();
const currentMapId = ref<string | null>(null);

function createWorldMapManager() {
  const loadJSON = (path: string) => fetch(path).then(res => res.json());

  const loadMapAndCollision = async (mapPath: string): Promise<void> => {
    try {
      // 1. Carregar o Manifesto Raiz
      const manifest: MapManifest = await loadJSON(mapPath);
      
      // 2. Carregar todas as partes do mapa
      const [visualData, audioData, playerData, eventsData, collisionData] = await Promise.all([
        loadJSON(manifest.visual),
        loadJSON(manifest.audio),
        loadJSON(manifest.player),
        loadJSON(manifest.events),
        loadJSON(manifest.collisionMap),
      ]);

      // 3. Mesclar em um único objeto MapDefinition
      const mapData: MapDefinition = {
        ...visualData,
        ...audioData,
        ...playerData,
        collisionMap: manifest.collisionMap,
        eventPlacements: eventsData.placements as EventPlacement[], // Assume que o events.json tem uma propriedade 'placements'
      };
      
      mapCache.set(mapPath, mapData);
      collisionCache.set(manifest.collisionMap, collisionData);

    } catch (error) {
      console.error(`[WorldMapManager] Failed to load map: ${mapPath}`, error);
      throw error;
    }
  };

  const setCurrentMap = (mapPath: string) => {
    if (mapCache.has(mapPath)) {
      currentMapId.value = mapPath;
    } else {
      console.error(`[WorldMapManager] Map not preloaded: ${mapPath}`);
    }
  };
  
  const mapData = computed(() => currentMapId.value ? mapCache.get(currentMapId.value) : null);
  const collisionData = computed(() => {
    const map = mapData.value;
    return map ? collisionCache.get(map.collisionMap) : null;
  });

  return { loadMapAndCollision, setCurrentMap, mapData, collisionData };
}

let instance: ReturnType<typeof createWorldMapManager>;

export function useWorldMapManager() {
  if (!instance) {
    instance = createWorldMapManager();
  }
  return instance;
}

import { ref, onUnmounted } from 'vue';

export function useTypewriter() {
  const displayedText = ref('');
  const isFinished = ref(true);
  let intervalId: number | null = null;

  const stop = () => {
    // Usamos window.clearInterval e garantimos que intervalId seja um number ou null
    if (intervalId !== null) {
      window.clearInterval(intervalId);
      intervalId = null;
    }
    isFinished.value = true;
  };

  const start = (fullText: string, speed?: number) => {
    stop();
    displayedText.value = '';
    
    if (!speed || speed <= 0) {
      displayedText.value = fullText;
      isFinished.value = true;
      return;
    }

    isFinished.value = false;
    let currentIndex = 0;
    
    // Usamos window.setInterval para garantir que o tipo de retorno seja 'number' no contexto do navegador
    intervalId = window.setInterval(() => {
      if (currentIndex < fullText.length) {
        displayedText.value += fullText[currentIndex];
        currentIndex++;
      } else {
        stop();
      }
    }, speed);
  };

  const skip = (fullText: string) => {
    stop();
    displayedText.value = fullText;
  };

  onUnmounted(stop);

  return {
    displayedText,
    isFinished,
    start,
    skip,
  };
}

interface CutsceneUIState {
  isActive: boolean;
  dialogueLine: string | null;
  typewriterSpeed: number | undefined;
  currentDialogueStyle: DialogueStyleConfig;
  backgroundColor: string;
  backgroundImage: string | null;
  overlayOpacity: number;
  actors: ActorState[];
}

export const useCutsceneUIStore = defineStore('cutscene-ui', () => {
  const state = reactive<CutsceneUIState>({
    isActive: false,
    dialogueLine: null,
    typewriterSpeed: undefined,
    currentDialogueStyle: {},
    backgroundColor: 'transparent',
    backgroundImage: null,
    overlayOpacity: 0,
    actors: [],
  });

  function showOverlay() { state.isActive = true; }
  function hideOverlay() { state.isActive = false; }

  function setDialogue(line: string | null, speed?: number) {
    state.dialogueLine = line;
    state.typewriterSpeed = speed;
  }

  function setDialogueStyle(style: DialogueStyleConfig) { state.currentDialogueStyle = style; }
  function setBackground(config: { color?: string; image?: string }) {
    state.backgroundColor = config.color || 'transparent';
    state.backgroundImage = config.image || null;
  }
  function setOverlayOpacity(opacity: number) { state.overlayOpacity = opacity; }
  
  function addActor(actorData: Omit<ActorState, 'transitionDuration'>) {
    const newActor: ActorState = {
      ...actorData,
      transitionDuration: 500 // Duração padrão
    };
    state.actors = state.actors.filter(a => a.id !== newActor.id);
    state.actors.push(newActor);
  }
  
  function removeActor(actorId: string) {
    state.actors = state.actors.filter(a => a.id !== actorId);
  }

  function setActorOpacity(actorId: string, opacity: number) {
    const actor = state.actors.find(a => a.id === actorId);
    if (actor) {
      actor.opacity = opacity;
    }
  }

  function setActorTransitionDuration(actorId: string, duration: number) {
    const actor = state.actors.find(a => a.id === actorId);
    if (actor) {
      actor.transitionDuration = duration;
    }
  }

  function reset() {
    state.isActive = false;
    state.dialogueLine = null;

import { defineStore } from 'pinia';
import { ref } from 'vue'; // 'reactive' foi removido desta linha
import type { DialogueStyleConfig } from '@/types/Cutscene';

interface DialogueLine {
  text: string;
  speed?: number;
}

export const useDialogueStore = defineStore('dialogue', () => {
  const isActive = ref(false);
  const lines = ref<DialogueLine[]>([]);
  const currentIndex = ref(0);
  const dialogueStyle = ref<DialogueStyleConfig>({});

  function start(dialogueLines: string[], typewriterSpeed?: number, style?: DialogueStyleConfig) {
    lines.value = dialogueLines.map(line => ({
      text: line,
      speed: typewriterSpeed,
    }));
    currentIndex.value = 0;
    isActive.value = true;
    dialogueStyle.value = style || {};
  }

  function next(): boolean {
    if (currentIndex.value < lines.value.length - 1) {
      currentIndex.value++;
      return true;
    }
    return false;
  }

  function end() {
    isActive.value = false;
    lines.value = [];
    currentIndex.value = 0;
    dialogueStyle.value = {};
  }

  function getCurrentLine(): string {
    return lines.value[currentIndex.value]?.text || '';
  }
  
  function getCurrentSpeed(): number | undefined {
    return lines.value[currentIndex.value]?.speed;
  }

  return { isActive, currentIndex, dialogueStyle, start, next, end, getCurrentLine, getCurrentSpeed };
});

export const useSwitchesStore = defineStore('switches', () => {
  // O estado é um objeto reativo para armazenar qualquer flag do jogo.
  const state = reactive<Record<string, any>>({});

  /**
   * Define ou atualiza o valor de um switch (flag) no estado global do jogo.
   * @param key O identificador único para o switch (ex: 'burro_chosen').
   * @param value O valor a ser armazenado (geralmente boolean, mas pode ser qualquer tipo).
   */
  function setSwitch(key: string, value: any) {
    state[key] = value;
    console.log(`[SwitchesStore] Switch set: '${key}' =`, value);
  }

  /**
   * Obtém o valor de um switch do estado global.
   * @param key O identificador do switch a ser verificado.
   * @returns O valor do switch, ou `undefined` se ele nunca foi definido.

  function getSwitch(key: string): any {
    return state[key];
  }

  return { state, setSwitch, getSwitch };
